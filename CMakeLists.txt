cmake_minimum_required(VERSION 3.20)
project(Tiaoyin VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(Threads REQUIRED)
add_library(tiaoyin_core src/audio_core.cpp src/recorder.cpp)
target_include_directories(tiaoyin_core PUBLIC src)
target_link_libraries(tiaoyin_core PUBLIC Threads::Threads)
if(WIN32)
  enable_language(RC)
  add_executable(Tiaoyin WIN32 src/main_win.cpp src/wasapi_engine.cpp src/app.rc)
  target_compile_definitions(Tiaoyin PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(Tiaoyin PRIVATE tiaoyin_core ole32 uuid avrt comctl32 comdlg32 shell32 propsys)
  if(MSVC)
    target_link_options(Tiaoyin PRIVATE /MANIFEST:NO)
  elseif(MINGW)
    target_link_options(Tiaoyin PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(tiaoyin_tests tests/core_tests.cpp)
  target_link_libraries(tiaoyin_tests PRIVATE tiaoyin_core)
  add_test(NAME core COMMAND tiaoyin_tests)
endif()
